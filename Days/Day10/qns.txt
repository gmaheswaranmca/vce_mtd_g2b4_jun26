class Solution:
    def maxLevelSum(self, root: Optional[TreeNode]) -> int:
        queue = deque()
        if root != None:
            queue.append(root)
            maxLevel = 0
            maxSum = -1*(10**5)
        level = 0  
        while queue:
            size = len(queue)
            s = 0
            for I in range(size):
                node = queue.popleft()
                s += node.val
                if node.left != None:
                    queue.append(node.left)
                if node.right != None:
                    queue.append(node.right)
            level += 1
            if s > maxSum:
                maxSum = s
                maxLevel = level
        return maxLevel
