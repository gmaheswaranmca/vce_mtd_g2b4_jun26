    stk.push(source)
    visited[source] = true
    while stk is not empty
        v = stk.pop()
        if v == destination
            return true

findCircleNum(isConnected)
    n = len(isConnected)
    visited = []
    for i <- 0 .. n-1
        visited.add(false) 
    p = 0
    for i <- 0..n-1
        if !visited[i]
            p++
            dfs(i)
    return p
